% Prepare the title's number without placing an anchor outside the box.
% The before-upper hook runs once and puts the reference in the box's content,
% so both ordinary body labels and links survive page breaks correctly.
\newenvironment{bigidea}[1][]{%
  \stepcounter{bigidea}%
  \begin{tcolorbox}[enhanced,breakable,boxrule=0pt,arc=1mm,
    colback=bookGold!10!ctpBase,coltitle=bookInk,colupper=bookInk,
    colbacktitle=bookGold!22!ctpBase,colframe=ctpSurface1,
    fonttitle=\sffamily\bfseries,
    before upper={\addtocounter{bigidea}{-1}\refstepcounter{bigidea}\RaggedRight},
    title={Big Idea \thebigidea\ifstrempty{#1}{}{ --- #1}},
    code={\ifstrempty{#1}{\tcbset{nameref={Big Idea}}}{\tcbset{nameref={#1}}}},
    left=3mm,right=3mm,top=2mm,bottom=2mm,before skip=12pt,after skip=12pt]
}{\end{tcolorbox}}
