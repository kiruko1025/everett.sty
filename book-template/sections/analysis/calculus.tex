\section{Aligned calculations}
Use an \texttt{align} environment to align related equalities. A label belongs
on the particular line that you want to reference:
\begin{align}
  \int_0^1 (x+x^2)\,\dd x
    &=\left[\frac{x^2}{2}+\frac{x^3}{3}\right]_0^1 \notag\\
    &=\frac12+\frac13=\frac56.\label{eq:polynomial-integral}
\end{align}
Equation~\eqref{eq:polynomial-integral} uses the local macro \verb|\dd| for
an upright differential. Local macros are defined in
\texttt{macros/macros.tex}; they are not part of \texttt{macro.sty}.

For a function of two variables, the package shorthand
\verb|\fpartial{f}{x}| gives
\[
\begin{aligned}
  f(x,y)&=x^2y+\sin y,\\
  \fpartial{f}{x}&=2xy,
  &\fpartial{f}{y}&=x^2+\cos y.
\end{aligned}
\]

\section{A basic continuity proof}
\begin{theorem}\label{thm:square-continuous}
The function $f:\RR\to\RR$ defined by $f(x)=x^2$ is continuous at every
$a\in\RR$.
\end{theorem}
\begin{proof}
Let $\eps>0$, and choose
\[
  \delta=\min\left\{1,\frac{\eps}{2\abs{a}+1}\right\}.
\]
If $\abs{x-a}<\delta$, then $\abs{x+a}\le\abs{x-a}+2\abs{a}<1+2\abs{a}$.
Consequently,
\[
  \abs{x^2-a^2}=\abs{x-a}\abs{x+a}
  <\delta(1+2\abs{a})\le\eps.
\]
This is the epsilon-delta condition for continuity at $a$.
\end{proof}

\begin{intuition}
The restriction $\delta\le1$ first bounds the factor $\abs{x+a}$. The other
restriction then makes the product as small as desired.
\end{intuition}

\begin{exercise}[A cubic estimate]\label{ex:cubic-estimate}
Adapt the proof of \cref{thm:square-continuous} to show that $x\mapsto x^3$
is continuous. Begin with $x^3-a^3=(x-a)(x^2+ax+a^2)$.
\end{exercise}
